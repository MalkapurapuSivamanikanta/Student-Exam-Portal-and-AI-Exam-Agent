-- 1. students_info
CREATE TABLE students_info (
    student_id          SERIAL          PRIMARY KEY,
    student_name        VARCHAR(100)    NOT NULL,
    gender              CHAR(1),
    id_card_no          VARCHAR(30)     NOT NULL,
    student_course      VARCHAR(20)     NOT NULL,   -- e.g. 'B.Tech'
    student_branch      VARCHAR(50)     NOT NULL,   -- e.g. 'CSE'
    chosen_subject      VARCHAR(100),               -- e.g. 'Machine Learning'
    section             VARCHAR(2)      NOT NULL,
    current_year        SMALLINT,
    written_exam        SMALLINT        NOT NULL DEFAULT 0,   -- 0 = not written, 1 = written
    exam_score          NUMERIC(4,2),
    created_at          TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT uq_students_id_card_no   UNIQUE (id_card_no),
    CONSTRAINT chk_gender               CHECK (gender IN ('M', 'F', 'O')),
    CONSTRAINT chk_current_year         CHECK (current_year BETWEEN 1 AND 4),
    CONSTRAINT chk_written_exam         CHECK (written_exam IN (0, 1)),
    CONSTRAINT chk_exam_score_range     CHECK (exam_score BETWEEN 0 AND 10),
    CONSTRAINT chk_exam_score_consistency
        CHECK (
            (written_exam = 1) OR (written_exam = 0 AND exam_score IS NULL)
        )
);

CREATE INDEX idx_students_course_branch_section
    ON students_info (student_course, student_branch, section);
CREATE INDEX idx_students_written_exam
    ON students_info (written_exam);

-- 3. student_login
CREATE TABLE student_login (
    login_id        SERIAL          PRIMARY KEY,
    student_id      INT             NOT NULL,
    login_password  VARCHAR(255)    NOT NULL,   -- plain text for testing only; use a HASH in production
    is_active       BOOLEAN         NOT NULL DEFAULT TRUE,
    last_login_at   TIMESTAMP,
    created_at      TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_login_student
        FOREIGN KEY (student_id) REFERENCES students_info (student_id)
        ON DELETE CASCADE,
    CONSTRAINT uq_login_student_id UNIQUE (student_id)   -- one login per student
);

Student portal flow:
1. The student logs in with their ID card number and password.
2. The portal verifies the credentials and displays the subject from the student record.
3. Starting the exam fetches the student's exam details and generates questions for that subject.
4. Submitting the exam saves the score and written-exam status to students_info.
5. Students whose written_exam status is already set cannot start another attempt; the portal shows their saved score.
